SQL-скрипт таблиц для Supabase 
create table if not exists proplan_users (
  id text primary key, name text not null, email text unique not null,
  pass_hash text not null, role text not null, created_at timestamptz default now()
);
create table if not exists proplan_teams (
  id text primary key, name text not null, category text default '',
  invite_code text unique not null, admin_id text not null,
  members jsonb not null default '[]'::jsonb, updated_at timestamptz default now()
);
create table if not exists proplan_team_data (
  team_id text primary key, plan jsonb not null default '{}'::jsonb,
  monitoring jsonb not null default '{}'::jsonb, exercises jsonb not null default '[]'::jsonb,
  sessions jsonb not null default '[]'::jsonb, updated_at timestamptz default now()
);
alter table proplan_users enable row level security;
alter table proplan_teams enable row level security;
alter table proplan_team_data enable row level security;
create policy "all_u" on proplan_users for all using (true) with check (true);
create policy "all_t" on proplan_teams for all using (true) with check (true);
create policy "all_d" on proplan_team_data for all using (true) with check (true);
